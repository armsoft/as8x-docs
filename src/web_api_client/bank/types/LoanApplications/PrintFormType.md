Հայտի տպվող ձևի տեսակ։ Enum-ը նախատեսված է հայտի ստեղծման համար։

```c#
/// <summary> Հայտի տպվող ձևի տեսակ </summary>
public enum PrintFormType
{
    /// <summary> Պայմանագիր  </summary>    
    Contract = 1,

    /// <summary> Արբիտրաժային համաձայնագիր </summary>    
    Arbitrage = 2,
    
    /// <summary> Անհատական թերթիկ </summary>    
    IndividualSheet = 3
}
```
